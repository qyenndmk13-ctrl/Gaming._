    ctx.globalAlpha=.12;

    ctx.globalAlpha=1;

  }
 }
}

function isWall(x,y){
 let s=player.size;
 let pts=[[x,y],[x+s,y],[x,y+s],[x+s,y+s]];

 return pts.some(([px,py])=>{
  return maps[currentMap].rows[Math.floor(py/TILE)]?.[Math.floor(px/TILE)]==="#";
 });
}

function movePlayer(dt){
 if(gameOver||won)return;

 let dx=0,dy=0;

 if(keys.ArrowUp||keys.w)dy=-1;
 if(keys.ArrowDown||keys.s)dy=1;
 if(keys.ArrowLeft||keys.a)dx=-1;
 if(keys.ArrowRight||keys.d)dx=1;

 if(dx&&dy){dx*=.707;dy*=.707;}

 if(dx>0)player.direction="right";
 if(dx<0)player.direction="left";
 if(dy>0)player.direction="down";
 if(dy<0)player.direction="up";

 let speed=player.speed*dt;

 if(!isWall(player.x+dx*speed,player.y))player.x+=dx*speed;
 if(!isWall(player.x,player.y+dy*speed))player.y+=dy*speed;
}

function damageMonster(m,damage){
 m.hp-=damage;

 effects.push({
  x:m.x+15,y:m.y+15,
  radius:24,color:"#facc15",
  life:12,maxLife:12
 });

 if(m.hp<=0){
  coins+=m.boss?100+level*30:10;
  exp+=m.boss?100:20;
  if(m.boss){
   bossAlive=false;
   say("BOSS DIKALAHKAN! +BONUS!");
  }else{
   say("Monster dikalahkan!");
  }
  return true;
 }
 return false;
}

function attack(){
 if(gameOver||won||attackCooldown>0)return;

 attackCooldown=20;

 monsters=monsters.filter(m=>{
  let dx=player.x-m.x;
  let dy=player.y-m.y;
  let dist=Math.hypot(dx,dy);

  let facing=
   (player.direction==="right"&&m.x>=player.x)||
   (player.direction==="left"&&m.x<=player.x)||
   (player.direction==="up"&&m.y<=player.y)||
   (player.direction==="down"&&m.y>=player.y);

  if(dist<80&&facing){
   return !damageMonster(m,atk);
  }
  return true;
 });
}

function castFire(){
 if(gameOver||won||skillCooldown.fire>0)return;
 if(mp<25){say("MP tidak cukup!");return;}

 mp-=25;
 skillCooldown.fire=90;

  x:player.x+11,y:player.y+11,
  radius:95,color:"#fb923c",
  life:25,maxLife:25
 });

 monsters=monsters.filter(m=>{
  let dist=Math.hypot(player.x-m.x,player.y-m.y);
  if(dist<120)return !damageMonster(m,atk*2+15);
  return true;
 });

 say("🔥 FIREBALL!");
}

function castThunder(){
 if(gameOver||won||skillCooldown.thunder>0)return;

  return true;

}

function castHeal(){

 if(mp<30){say("MP tidak cukup!");return;}
 if(hp>=maxHp){say("HP masih penuh!");return;}

 mp-=30;
 hp=Math.min(maxHp,hp+45);
 skillCooldown.heal=180;

 effects.push({
  x:player.x+11,y:player.y+11,
  radius:65,color:"#4ade80",
  life:30,maxLife:30
 });

 say("💚 HEALING MAGIC!");
}

function updateMonsters(dt){
 monsters.forEach(m=>{
  let dx=player.x-m.x;
  let dy=player.y-m.y;
  let dist=Math.hypot(dx,dy);

  if(dist<270&&dist>1){
   let nx=m.x+dx/dist*m.speed*dt;
   let ny=m.y+dy/dist*m.speed*dt;

   let tx=Math.floor(nx/TILE);
   let ty=Math.floor(ny/TILE);

   if(maps[currentMap].rows[ty]?.[tx]!== "#"){
    m.x=nx;
    m.y=ny;
   }
  }

  if(dist<(m.boss?47:27)){
   hp-=Math.max(.02,m.damage-armor*.015)*dt;

   if(hp<=0){
    hp=0;
    gameOver=true;
    say("GAME OVER! Petualangan berakhir.");
   }
  }
 });
}

function collectTreasures(){
 treasures.forEach(t=>{
  if(!t.collected&&Math.hypot(player.x-t.x,player.y-t.y)<30){
   t.collected=true;
   coins+=10;
   say("💰 Treasure ditemukan! +10 Coin");
  }
 });
}

function checkExit(){
 let dist=Math.hypot(player.x-exit.x,player.y-exit.y);

 if(dist<30){
  if(bossAlive){
   say("Kalahkan BOSS sebelum membuka portal!");
   return;
  }

  if(level<5){
   level++;
   hp=Math.min(maxHp,hp+25);
   mp=Math.min(maxMp,mp+35);
   createLevel();
  }else{
   won=true;
   say("🏆 KAMU MENAKLUKKAN SEMUA DUNGEON!");
  }
 }
}

function updateEffects(dt){
 effects.forEach(e=>e.life-=dt);
 effects=effects.filter(e=>e.life>0);

 attackCooldown=Math.max(0,attackCooldown-dt);

 for(let k in skillCooldown){
  skillCooldown[k]=Math.max(0,skillCooldown[k]-dt);
 }

 mp=Math.min(maxMp,mp+.025*dt);
}

  }

}

function draw(){

 drawExit();
 drawTreasures();
 drawMonsters();

 drawMiniMap();

 if(gameOver||won){
  ctx.fillStyle="rgba(0,0,0,.85)";
  ctx.fillRect(0,0,640,480);

  ctx.textAlign="center";
  ctx.fillStyle=won?"#facc15":"#ef4444";
  ctx.font="bold 34px monospace";
  ctx.fillText(won?"VICTORY!":"GAME OVER",320,200);

  ctx.fillStyle="white";
  ctx.font="17px monospace";
  ctx.fillText("Dungeon: "+level+"/5",320,245);
  ctx.fillText("Coins: "+coins,320,275);
  ctx.fillText("EXP: "+exp,320,305);
  ctx.textAlign="left";
 }

 ctx.fillStyle="white";
 ctx.font="14px monospace";
 ctx.fillText(maps[currentMap].name,10,20);
}

function loop(time=0){
 let dt=Math.min((time-lastTime)/16.67,2);
 lastTime=time;

 if(!gameOver&&!won){
  movePlayer(dt);
  updateMonsters(dt);
  collectTreasures();
  checkExit();
  updateEffects(dt);
  updateUI();
 }

 draw();
 requestAnimationFrame(loop);
}

// KEYBOARD
document.addEventListener("keydown",e=>{
 let k=e.key.toLowerCase();

 if(["arrowup","arrowdown","arrowleft","arrowright"," "].includes(k))
 e.preventDefault();

 keys[k]=true;

 if(k===" "&&!e.repeat)attack();
 if(k==="1"&&!e.repeat)castFire();
 if(k==="2"&&!e.repeat)castThunder();
 if(k==="3"&&!e.repeat)castHeal();
 if(k==="p"&&!e.repeat)usePotion();
});

document.addEventListener("keyup",e=>{
 keys[e.key.toLowerCase()]=false;
});

// MOBILE
document.querySelectorAll("[data-key]").forEach(btn=>{
 let k=btn.dataset.key;

 btn.addEventListener("pointerdown",e=>{
  e.preventDefault();
  keys[k]=true;
 });

 ["pointerup","pointerleave","pointercancel"].forEach(ev=>{
  btn.addEventListener(ev,()=>keys[k]=false);
 });
});
